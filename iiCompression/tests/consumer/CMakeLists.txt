cmake_minimum_required(VERSION 3.24)
project(iiCompressionInstalledConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 ON)
find_package(iiCompression 0.1 CONFIG REQUIRED)
if(TARGET Qt5::Core OR TARGET Qt6::Core)
    message(FATAL_ERROR "The compression package imported Qt")
endif()
add_executable(gzip-consumer ../gzip.cpp)
target_link_libraries(gzip-consumer PRIVATE iiCompression::iiCompression)
enable_testing()
add_test(NAME iiCompression.InstalledGzip COMMAND gzip-consumer)
