cmake_minimum_required(VERSION 3.24)
project(iiSDKPerformance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
get_filename_component(SDK_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(SDK_PERFORMANCE_LEGACY "Also benchmark existing Qt-based catalog and LVRS APIs" OFF)
set(SDK_PERFORMANCE_BASELINE "" CACHE PATH "Pinned source tree containing before-change module/path files")
set(SDK_PERFORMANCE_BASELINE_CORE "" CACHE FILEPATH "Before-change iiSDKCore library")

find_package(iiSDKCore CONFIG REQUIRED)
find_package(iiColorManager CONFIG REQUIRED)
find_package(iiPaintEngine CONFIG REQUIRED)
find_package(iiGraphicAdjust CONFIG REQUIRED)
find_package(iiFileProviderNative CONFIG REQUIRED)

add_executable(sdk_runtime_benchmark runtime.cpp)
target_link_libraries(sdk_runtime_benchmark PRIVATE iiSDKCore::iiSDKCore iiColorManager::iiColorManager)
add_executable(sdk_filter_benchmark filters.cpp)
target_link_libraries(sdk_filter_benchmark PRIVATE iiPaintEngine::iiPaintEngine iiGraphicAdjust::iiGraphicAdjust)
add_executable(sdk_authorship_benchmark authorship.cpp)
target_link_libraries(sdk_authorship_benchmark PRIVATE iiFileProvider::Native)
foreach(target sdk_runtime_benchmark sdk_filter_benchmark sdk_authorship_benchmark)
    add_test(NAME ${target}.smoke COMMAND ${target})
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/linked-native-packages-$<CONFIG>.txt"
    CONTENT "iiSDKCore=$<TARGET_FILE:iiSDKCore::iiSDKCore>\niiColorManager=$<TARGET_FILE:iiColorManager::iiColorManager>\niiPaintEngine=$<TARGET_FILE:iiPaintEngine::iiPaintEngine>\niiGraphicAdjust=$<TARGET_FILE:iiGraphicAdjust::iiGraphicAdjust>\niiFileProviderNative=$<TARGET_FILE:iiFileProvider::Native>\n")

if(SDK_PERFORMANCE_LEGACY)
    find_package(Qt6 REQUIRED COMPONENTS Core Qml)
    find_package(iiLocalLLM CONFIG REQUIRED)
    find_package(LVRS CONFIG REQUIRED)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/linked-legacy-packages-$<CONFIG>.txt"
        CONTENT "iiLocalLLM=$<TARGET_FILE:iiLocalLLM::iiLocalLLM>\nLVRS=$<TARGET_FILE:LVRS::LVRS>\n")
    add_executable(sdk_catalog_benchmark catalog.cpp)
    target_link_libraries(sdk_catalog_benchmark PRIVATE iiLocalLLM::iiLocalLLM Qt6::Core)
    add_executable(sdk_hierarchy_benchmark hierarchy.cpp "${SDK_SOURCE_ROOT}/LVRS/src/backend/hierarchy/hierarchyvisibility.cpp")
    target_include_directories(sdk_hierarchy_benchmark PRIVATE "${SDK_SOURCE_ROOT}/LVRS/src")
    target_link_libraries(sdk_hierarchy_benchmark PRIVATE Qt6::Core)
    add_executable(sdk_table_benchmark table.cpp)
    target_include_directories(sdk_table_benchmark PRIVATE "${SDK_SOURCE_ROOT}/LVRS/src" "${SDK_SOURCE_ROOT}/LVRS/build")
    target_link_libraries(sdk_table_benchmark PRIVATE LVRS::LVRS Qt6::Core Qt6::Qml)
    foreach(target sdk_catalog_benchmark sdk_hierarchy_benchmark sdk_table_benchmark)
        add_test(NAME ${target}.smoke COMMAND ${target})
    endforeach()
endif()

# Baseline targets are development oracles, never installed SDK targets.
# Public signatures/layouts used by these probes are unchanged by this refactor.
if(SDK_PERFORMANCE_BASELINE)
    if(NOT APPLE OR NOT EXISTS "${SDK_PERFORMANCE_BASELINE_CORE}")
        message(FATAL_ERROR "The pinned-source oracle currently requires macOS and a separately built baseline Core library")
    endif()
    set(before "${SDK_PERFORMANCE_BASELINE}")
    get_filename_component(before_core_dir "${SDK_PERFORMANCE_BASELINE_CORE}" DIRECTORY)
    add_executable(sdk_runtime_benchmark_before runtime.cpp "${before}/iiColorManager/src/ColorManager.cpp")
    target_include_directories(sdk_runtime_benchmark_before BEFORE PRIVATE "${before_core_dir}/include" "${SDK_SOURCE_ROOT}/iiColorManager/include" "${SDK_SOURCE_ROOT}/iiColorManager/build/include")
    target_include_directories(sdk_runtime_benchmark_before PRIVATE "$<TARGET_PROPERTY:iiSDKCore::iiSDKCore,INTERFACE_INCLUDE_DIRECTORIES>")
    target_link_libraries(sdk_runtime_benchmark_before PRIVATE "${SDK_PERFORMANCE_BASELINE_CORE}")

    add_executable(sdk_filter_benchmark_before filters.cpp "${before}/iiPaintEngine/src/Filter/FilterPipeline.cpp"
        "${before}/iiGraphicAdjust/src/iiGraphicAdjust/BlurFilters.cpp" "${before}/iiGraphicAdjust/src/iiGraphicAdjust/PixelFilters.cpp")
    target_include_directories(sdk_filter_benchmark_before PRIVATE "${SDK_SOURCE_ROOT}/iiPaintEngine/src/Filter" "${SDK_SOURCE_ROOT}/iiGraphicAdjust/src/iiGraphicAdjust")
    target_link_libraries(sdk_filter_benchmark_before PRIVATE iiPaintEngine::iiPaintEngine iiGraphicAdjust::iiGraphicAdjust)

    add_executable(sdk_authorship_benchmark_before authorship.cpp "${before}/iiFileProvider/Native/src/Authorship.cpp"
        "${SDK_SOURCE_ROOT}/iiFileProvider/Native/src/Sha256.cpp" "${SDK_SOURCE_ROOT}/iiFileProvider/Native/third_party/utf8proc/utf8proc.c")
    target_compile_definitions(sdk_authorship_benchmark_before PRIVATE UTF8PROC_STATIC)
    target_include_directories(sdk_authorship_benchmark_before PRIVATE "${SDK_SOURCE_ROOT}/iiFileProvider/Native/src"
        "${SDK_SOURCE_ROOT}/iiFileProvider/Native/third_party" "${SDK_SOURCE_ROOT}/iiFileProvider/Native/third_party/utf8proc"
        "$<TARGET_PROPERTY:iiFileProvider::Native,INTERFACE_INCLUDE_DIRECTORIES>")
    if(SDK_PERFORMANCE_LEGACY)
        add_executable(sdk_catalog_benchmark_before catalog.cpp "${before}/iiLocalLLM/src/Parameters.cpp")
        target_include_directories(sdk_catalog_benchmark_before PRIVATE "${SDK_SOURCE_ROOT}/iiLocalLLM/src")
        target_link_libraries(sdk_catalog_benchmark_before PRIVATE iiLocalLLM::iiLocalLLM Qt6::Core)
        add_executable(sdk_hierarchy_benchmark_before hierarchy.cpp "${before}/LVRS/src/backend/hierarchy/hierarchyvisibility.cpp")
        target_include_directories(sdk_hierarchy_benchmark_before PRIVATE "${SDK_SOURCE_ROOT}/LVRS/src" "${SDK_SOURCE_ROOT}/LVRS/src/backend/hierarchy")
        target_link_libraries(sdk_hierarchy_benchmark_before PRIVATE Qt6::Core)
        add_executable(sdk_table_benchmark_before table.cpp "${before}/LVRS/src/backend/model/tablemodel.cpp")
        target_include_directories(sdk_table_benchmark_before PRIVATE "${SDK_SOURCE_ROOT}/LVRS/src" "${SDK_SOURCE_ROOT}/LVRS/src/backend/model" "${SDK_SOURCE_ROOT}/LVRS/build")
        target_link_libraries(sdk_table_benchmark_before PRIVATE LVRS::LVRS Qt6::Core Qt6::Qml)
    endif()
endif()
