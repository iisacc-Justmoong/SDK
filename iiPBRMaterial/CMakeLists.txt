cmake_minimum_required(VERSION 3.23)
project(iiPBRMaterial VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(BUILD_SHARED_LIBS "Build SDK libraries as shared libraries" ON)
if(EMSCRIPTEN AND BUILD_SHARED_LIBS)
  set_property(GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS TRUE)
endif()
add_library(iiPBRMaterial src/CustomAtom.cpp src/CustomBond.cpp src/MolecularFormula.cpp src/Molecule.cpp src/MolecularMap.cpp src/Texture.cpp src/PBRMaterial.cpp src/OrganicMaterial.cpp)
add_library(iiPBRMaterial::iiPBRMaterial ALIAS iiPBRMaterial)
target_compile_features(iiPBRMaterial PUBLIC cxx_std_23)
target_include_directories(iiPBRMaterial PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/iiPBRMaterial>)
set_target_properties(iiPBRMaterial PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(BUILD_SHARED_LIBS)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set_target_properties(iiPBRMaterial PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
      MACOSX_FRAMEWORK_IDENTIFIER "com.iisacc.iiPBRMaterial")
  elseif(EMSCRIPTEN)
    target_compile_options(iiPBRMaterial PRIVATE -fPIC -sSIDE_MODULE=1)
    target_link_options(iiPBRMaterial PRIVATE -sSIDE_MODULE=1)
  elseif(APPLE)
    set_property(TARGET iiPBRMaterial APPEND PROPERTY INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_property(TARGET iiPBRMaterial APPEND PROPERTY INSTALL_RPATH "$ORIGIN")
  endif()
endif()
if(MSVC)
  target_compile_options(iiPBRMaterial PRIVATE /W4 /permissive-)
else()
  target_compile_options(iiPBRMaterial PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(iiPBRMaterialTests tests/PBRMaterialTests.cpp)
  target_link_libraries(iiPBRMaterialTests PRIVATE iiPBRMaterial::iiPBRMaterial)
  target_compile_definitions(iiPBRMaterialTests PRIVATE IIPBR_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/resources/fixtures/material-id.ppm")
  add_test(NAME iiPBRMaterialTests COMMAND iiPBRMaterialTests)
  if(NOT CMAKE_CROSSCOMPILING AND (NOT DEFINED BUILD_SHARED_LIBS OR BUILD_SHARED_LIBS))
    add_test(NAME iiPBRMaterial.SharedLink COMMAND ${CMAKE_COMMAND}
      "-DLIBRARY_TYPE=$<TARGET_PROPERTY:iiPBRMaterial,TYPE>" "-DLIBRARY=$<TARGET_FILE:iiPBRMaterial>"
      "-DIMAGE=$<TARGET_FILE:iiPBRMaterialTests>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/shared_link.cmake")
  endif()
  add_executable(iiPBRMoleculeTests tests/MoleculeTests.cpp)
  target_link_libraries(iiPBRMoleculeTests PRIVATE iiPBRMaterial::iiPBRMaterial)
  add_test(NAME iiPBRMoleculeTests COMMAND iiPBRMoleculeTests)
  add_executable(iiPBROrganicMaterialTests tests/OrganicMaterialTests.cpp)
  target_link_libraries(iiPBROrganicMaterialTests PRIVATE iiPBRMaterial::iiPBRMaterial)
  add_test(NAME iiPBROrganicMaterialTests COMMAND iiPBROrganicMaterialTests)
  add_executable(iiPBRCustomAtomTests tests/CustomAtomTests.cpp)
  target_link_libraries(iiPBRCustomAtomTests PRIVATE iiPBRMaterial::iiPBRMaterial)
  add_test(NAME iiPBRCustomAtomTests COMMAND iiPBRCustomAtomTests)
  add_executable(iiPBRCustomBondTests tests/CustomBondTests.cpp)
  target_link_libraries(iiPBRCustomBondTests PRIVATE iiPBRMaterial::iiPBRMaterial)
  add_test(NAME iiPBRCustomBondTests COMMAND iiPBRCustomBondTests)
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME iiPBRMolecularSvgXml COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_molecular_svg.py $<TARGET_FILE:iiPBRMoleculeTests> $<TARGET_FILE:iiPBRCustomAtomTests> $<TARGET_FILE:iiPBRCustomBondTests>)
    endif()
  endif()
endif()
option(IIPBR_BUILD_EXAMPLE "Build the in-memory formula example" OFF)
if(IIPBR_BUILD_EXAMPLE)
  add_executable(iiPBRMaterialExample example/formula.cpp)
  target_link_libraries(iiPBRMaterialExample PRIVATE iiPBRMaterial::iiPBRMaterial)
  add_executable(iiPBRMoleculeExample example/molecule.cpp)
  target_link_libraries(iiPBRMoleculeExample PRIVATE iiPBRMaterial::iiPBRMaterial)
endif()
install(TARGETS iiPBRMaterial EXPORT iiPBRMaterialTargets
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} FRAMEWORK DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/CustomAtom.h src/CustomBond.h src/MolecularFormula.h src/Molecule.h src/MolecularMap.h src/Texture.h src/PBRMaterial.h src/OrganicMaterial.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiPBRMaterial)
install(EXPORT iiPBRMaterialTargets NAMESPACE iiPBRMaterial:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiPBRMaterial)
configure_package_config_file(cmake/iiPBRMaterialConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/iiPBRMaterialConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiPBRMaterial)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/iiPBRMaterialConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/iiPBRMaterialConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/iiPBRMaterialConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiPBRMaterial)
