cmake_minimum_required(VERSION 3.31)
project(iiGPU VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(BUILD_SHARED_LIBS "Build shared libraries" OFF)
option(IIGPU_BUILD_RUNTIME "Build the scheduler-independent completion observer" ON)
option(IIGPU_BUILD_EXAMPLES "Build the GPU acceleration example" OFF)
option(IIGPU_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(IIGPU_WEBGPU AUTO CACHE STRING "WebGPU component: AUTO, ON, OFF")
set(IIGPU_METAL AUTO CACHE STRING "Metal: AUTO, ON, OFF")
set(IIGPU_VULKAN AUTO CACHE STRING "Vulkan: AUTO, ON, OFF")
include(cmake/DawnTarget.cmake)
iigpu_dawn_target(_dawn_key "${CMAKE_SYSTEM_NAME}" "${CMAKE_SYSTEM_PROCESSOR}" "${CMAKE_OSX_ARCHITECTURES}")
set(IIGPU_DAWN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/resources/dawn/${_dawn_key}" CACHE PATH "Target-native Dawn distribution")
foreach(_api METAL VULKAN WEBGPU)
    set_property(CACHE IIGPU_${_api} PROPERTY STRINGS AUTO ON OFF)
    if(NOT IIGPU_${_api} MATCHES "^(AUTO|ON|OFF)$")
        message(FATAL_ERROR "IIGPU_${_api} must be AUTO, ON or OFF")
    endif()
endforeach()
find_package(Threads REQUIRED)
foreach(_header "${CMAKE_CURRENT_SOURCE_DIR}/src/GPU.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/Runtime/Completion.h")
    get_filename_component(_name "${_header}" NAME)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/iiGPU/${_name}"
        CONTENT "#pragma once\n#include \"${_header}\"\n")
endforeach()
file(GLOB _webgpu_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/WebGPUComponent/*.h")
foreach(_header IN LISTS _webgpu_headers)
    get_filename_component(_name "${_header}" NAME)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/include/iiWebGPU/${_name}"
        CONTENT "#pragma once\n#include \"${_header}\"\n")
endforeach()
add_library(iiGPU src/GPU.cpp)
add_library(iiGPU::iiGPU ALIAS iiGPU)
target_compile_features(iiGPU PUBLIC cxx_std_23)
target_link_libraries(iiGPU PRIVATE Threads::Threads)
target_include_directories(iiGPU PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(iiGPU PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/Backends")
set(IIGPU_HAS_METAL FALSE)
set(IIGPU_HAS_VULKAN FALSE)
if(APPLE AND NOT IIGPU_METAL STREQUAL "OFF")
    enable_language(OBJCXX)
    target_sources(iiGPU PRIVATE src/Backends/Metal.mm)
    target_compile_options(iiGPU PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(iiGPU PRIVATE "-framework Metal" "-framework Foundation")
    target_compile_definitions(iiGPU PRIVATE IIGPU_HAS_METAL=1)
    set(IIGPU_HAS_METAL TRUE)
elseif(IIGPU_METAL STREQUAL "ON")
    message(FATAL_ERROR "Metal requires an Apple platform SDK")
endif()
if(NOT IIGPU_VULKAN STREQUAL "OFF")
    find_package(Vulkan 1.1 QUIET)
    if(Vulkan_FOUND)
        target_sources(iiGPU PRIVATE src/Backends/Vulkan.cpp)
        target_link_libraries(iiGPU PRIVATE Vulkan::Vulkan)
        if(BUILD_SHARED_LIBS)
            target_link_libraries(iiGPU INTERFACE "$<LINK_ONLY:Vulkan::Vulkan>")
        endif()
        target_compile_definitions(iiGPU PRIVATE IIGPU_HAS_VULKAN=1)
        set(IIGPU_HAS_VULKAN TRUE)
    elseif(IIGPU_VULKAN STREQUAL "ON")
        message(FATAL_ERROR "Vulkan 1.1 headers and loader are required")
    endif()
endif()
set(IIGPU_HAS_WEBGPU FALSE)
if(NOT IIGPU_WEBGPU STREQUAL "OFF")
    if(EXISTS "${IIGPU_DAWN_ROOT}/iiwebgpu-source.json")
        file(READ "${IIGPU_DAWN_ROOT}/iiwebgpu-source.json" _dawn_metadata)
        string(JSON _dawn_recorded_target ERROR_VARIABLE _dawn_json_error GET "${_dawn_metadata}" platform)
        if(_dawn_json_error OR NOT _dawn_recorded_target STREQUAL _dawn_key)
            message(FATAL_ERROR "Dawn distribution does not match target ${_dawn_key}: ${IIGPU_DAWN_ROOT}")
        endif()
    endif()
    if(NOT TARGET dawn::webgpu_dawn)
        find_package(Dawn CONFIG QUIET HINTS "${IIGPU_DAWN_ROOT}" NO_DEFAULT_PATH)
    endif()
    if(TARGET dawn::webgpu_dawn)
        set(IIGPU_HAS_WEBGPU TRUE)
        add_library(iiGPUWebGPU STATIC src/WebGPUComponent/Memory.cpp src/WebGPUComponent/WebGPU.cpp
            src/WebGPUComponent/Texture.cpp src/WebGPUComponent/Arena.cpp)
        add_library(iiGPU::WebGPU ALIAS iiGPUWebGPU)
        add_library(iiWebGPU::iiWebGPU ALIAS iiGPUWebGPU)
        target_compile_features(iiGPUWebGPU PUBLIC cxx_std_23)
        target_include_directories(iiGPUWebGPU PUBLIC
            $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
            $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
            $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
        target_link_libraries(iiGPUWebGPU PUBLIC dawn::webgpu_dawn)
        set_target_properties(iiGPUWebGPU PROPERTIES EXPORT_NAME WebGPU CXX_EXTENSIONS OFF
            POSITION_INDEPENDENT_CODE ON)
        if(MSVC)
            target_compile_options(iiGPUWebGPU PRIVATE /W4 /permissive-)
        else()
            target_compile_options(iiGPUWebGPU PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    elseif(IIGPU_WEBGPU STREQUAL "ON")
        message(FATAL_ERROR "WebGPU requires a target-native Dawn distribution at IIGPU_DAWN_ROOT")
    endif()
endif()
set_target_properties(iiGPU PROPERTIES CXX_EXTENSIONS OFF OBJCXX_STANDARD 23
    OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON
    WINDOWS_EXPORT_ALL_SYMBOLS ON VERSION ${PROJECT_VERSION} SOVERSION 0)
if(MSVC)
    target_compile_options(iiGPU PRIVATE /W4 PUBLIC /Zc:__cplusplus /utf-8)
else()
    target_compile_options(iiGPU PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(IIGPU_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "IIGPU_SANITIZE requires Clang/GCC address and undefined behavior sanitizers")
    endif()
    target_compile_options(iiGPU PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(iiGPU PUBLIC -fsanitize=address,undefined)
    if(IIGPU_HAS_WEBGPU)
        target_compile_options(iiGPUWebGPU PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(iiGPUWebGPU PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(IIGPU_BUILD_RUNTIME)
    add_library(iiGPURuntime src/Runtime/Completion.cpp)
    add_library(iiGPU::Runtime ALIAS iiGPURuntime)
    set_target_properties(iiGPURuntime PROPERTIES EXPORT_NAME Runtime CXX_EXTENSIONS OFF
        WINDOWS_EXPORT_ALL_SYMBOLS ON POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(iiGPURuntime PUBLIC iiGPU)
    install(TARGETS iiGPURuntime EXPORT iiGPURuntimeTargets
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(TARGETS iiGPU EXPORT iiGPUTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/GPU.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiGPU)
if(IIGPU_HAS_WEBGPU)
    install(TARGETS iiGPUWebGPU EXPORT iiGPUWebGPUTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(DIRECTORY src/WebGPUComponent/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiWebGPU FILES_MATCHING PATTERN "*.h")
    install(DIRECTORY "${IIGPU_DAWN_ROOT}/lib/" DESTINATION ${CMAKE_INSTALL_LIBDIR}/iiGPU/dawn/lib
        PATTERN "*.iisacc.json" EXCLUDE PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE)
    foreach(_headers dawn webgpu webgpu_upstream)
        install(DIRECTORY "${IIGPU_DAWN_ROOT}/include/${_headers}" DESTINATION ${CMAKE_INSTALL_LIBDIR}/iiGPU/dawn/include
            PATTERN "*.iisacc.json" EXCLUDE PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE)
    endforeach()
    install(DIRECTORY "${IIGPU_DAWN_ROOT}/licenses/" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiGPU/dawn
        PATTERN "*.iisacc.json" EXCLUDE PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE)
    install(FILES docs/WebGPU-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiGPU)
    install(FILES tools/dawn.lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/iiGPU)
    install(EXPORT iiGPUWebGPUTargets NAMESPACE iiGPU:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiGPU)
    configure_package_config_file(cmake/iiWebGPUConfig.cmake.in
        ${CMAKE_CURRENT_BINARY_DIR}/iiWebGPUConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiWebGPU)
    write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/iiWebGPUConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/iiWebGPUConfig.cmake
        ${CMAKE_CURRENT_BINARY_DIR}/iiWebGPUConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiWebGPU)
endif()
if(IIGPU_BUILD_RUNTIME)
    install(FILES src/Runtime/Completion.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiGPU)
endif()
set(_pkg ${CMAKE_INSTALL_LIBDIR}/cmake/iiGPU)
configure_package_config_file(cmake/iiGPUConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/iiGPUConfig.cmake INSTALL_DESTINATION ${_pkg})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/iiGPUConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT iiGPUTargets NAMESPACE iiGPU:: DESTINATION ${_pkg})
if(IIGPU_BUILD_RUNTIME)
    install(EXPORT iiGPURuntimeTargets NAMESPACE iiGPU:: DESTINATION ${_pkg})
endif()
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/iiGPUConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/iiGPUConfigVersion.cmake DESTINATION ${_pkg})
install(FILES README.md AGENT.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiGPU)
install(DIRECTORY docs example DESTINATION ${CMAKE_INSTALL_DATADIR}/iiGPU PATTERN build EXCLUDE
    PATTERN "*.iisacc.json" EXCLUDE PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE)
if(BUILD_TESTING)
    add_test(NAME iiGPU.production_defaults COMMAND "${CMAKE_COMMAND}"
        "-DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}" "-DBINARY_ROOT=${CMAKE_CURRENT_BINARY_DIR}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/production_defaults.cmake")
    add_test(NAME iiGPU.source_layout COMMAND "${CMAKE_COMMAND}"
        "-DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/source_layout.cmake")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME iiGPU.build_layout COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/webgpu/build_layout.py)
    if(IIGPU_HAS_WEBGPU)
        add_executable(iiGPUWebGPUMergeTests tests/webgpu_merge.cpp)
        target_link_libraries(iiGPUWebGPUMergeTests PRIVATE iiGPU::iiGPU iiGPU::WebGPU)
        add_test(NAME iiGPU.webgpu.merge COMMAND iiGPUWebGPUMergeTests)
        add_executable(iiGPUWebGPUMemoryTests tests/webgpu/memory.cpp)
        target_link_libraries(iiGPUWebGPUMemoryTests PRIVATE iiGPU::WebGPU Threads::Threads)
        add_test(NAME iiGPU.webgpu.memory COMMAND iiGPUWebGPUMemoryTests)
        add_executable(iiGPUWebGPUTests tests/webgpu/gpu.cpp)
        target_link_libraries(iiGPUWebGPUTests PRIVATE iiGPU::WebGPU)
        foreach(_case Footprint Capabilities Compute Render Validation Lifetime)
            add_test(NAME iiGPU.webgpu.${_case} COMMAND iiGPUWebGPUTests ${_case})
            set_tests_properties(iiGPU.webgpu.${_case} PROPERTIES TIMEOUT 45)
        endforeach()
        foreach(_header Result Memory Native Texture WebGPU Arena)
            file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/header-check/${_header}.cpp" CONTENT "#include <iiWebGPU/${_header}.h>\n")
            add_library(iiGPUWebGPUHeader${_header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header-check/${_header}.cpp")
            target_link_libraries(iiGPUWebGPUHeader${_header} PRIVATE iiGPU::WebGPU)
        endforeach()
        add_test(NAME iiGPU.webgpu.bootstrap COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/webgpu/bootstrap.py)
        add_test(NAME iiGPU.webgpu.target_selection COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/webgpu/dawn_targets.cmake)
        add_test(NAME iiGPU.webgpu.relocated_consumer COMMAND ${Python3_EXECUTABLE} -B
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/webgpu/relocation.py ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR} "$<CONFIG>"
            "${Vulkan_INCLUDE_DIR}" "${Vulkan_LIBRARY}")
        set_tests_properties(iiGPU.webgpu.relocated_consumer PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endif()
    add_executable(iiGPURetirementTests tests/retirement.cpp)
    target_link_libraries(iiGPURetirementTests PRIVATE iiGPU::iiGPU)
    add_test(NAME iiGPU.retirement COMMAND iiGPURetirementTests)
    set_tests_properties(iiGPU.retirement PROPERTIES TIMEOUT 10)
    if(NOT IIGPU_HAS_METAL AND NOT IIGPU_HAS_VULKAN)
        add_executable(iiGPUUnsupportedTests tests/unsupported.cpp)
        target_link_libraries(iiGPUUnsupportedTests PRIVATE iiGPU::iiGPU)
        add_test(NAME iiGPU.unsupported COMMAND iiGPUUnsupportedTests)
        set_tests_properties(iiGPU.unsupported PROPERTIES TIMEOUT 10)
    endif()
    add_executable(iiGPUTests tests/gpu.cpp)
    target_link_libraries(iiGPUTests PRIVATE iiGPU::iiGPU)
    foreach(_case contracts transfers dispatch lifecycle concurrency interop)
        foreach(_api metal vulkan)
            add_test(NAME iiGPU.${_api}.${_case} COMMAND iiGPUTests ${_api} ${_case})
            set_tests_properties(iiGPU.${_api}.${_case} PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
        endforeach()
    endforeach()
    if(IIGPU_BUILD_RUNTIME)
        add_executable(iiGPUCompletionTests tests/completion_contracts.cpp)
        target_link_libraries(iiGPUCompletionTests PRIVATE iiGPU::Runtime Threads::Threads)
        add_test(NAME iiGPU.runtime COMMAND iiGPUCompletionTests)
        set_tests_properties(iiGPU.runtime PROPERTIES TIMEOUT 30)
    endif()
    add_test(NAME iiGPU.boundary COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/boundary.py ${CMAKE_CURRENT_SOURCE_DIR})
endif()
if(IIGPU_BUILD_EXAMPLES)
    add_executable(iiGPUAcceleration example/Acceleration/main.cpp)
    target_link_libraries(iiGPUAcceleration PRIVATE iiGPU::iiGPU)
    if(IIGPU_HAS_WEBGPU)
        add_executable(iiGPUWebGPUCompute example/WebGPU/compute.cpp)
        target_link_libraries(iiGPUWebGPUCompute PRIVATE iiGPU::WebGPU)
        if(BUILD_TESTING)
            add_test(NAME iiGPU.webgpu.example COMMAND iiGPUWebGPUCompute)
        endif()
    endif()
endif()
message(STATUS "iiGPU backends: Metal=${IIGPU_HAS_METAL} Vulkan=${IIGPU_HAS_VULKAN} WebGPU=${IIGPU_HAS_WEBGPU}")
