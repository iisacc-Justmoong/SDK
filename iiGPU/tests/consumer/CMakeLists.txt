cmake_minimum_required(VERSION 3.31)
project(iiGPUInstalledConsumer LANGUAGES CXX)
include(CTest)
option(WITH_RUNTIME "Verify the independent completion observer" OFF)
option(WITH_WEBGPU_MEMORY "Verify WebGPU headers and memory contracts without GPU execution" OFF)
option(WITH_CORE_COMPOSITION "Build the application-level Core/GPU composition" OFF)
if(WITH_WEBGPU_MEMORY)
    find_package(iiGPU 0.1 CONFIG REQUIRED COMPONENTS WebGPU)
    find_package(iiWebGPU 0.1 CONFIG REQUIRED)
    add_executable(consumer ../webgpu_merge.cpp)
    target_link_libraries(consumer PRIVATE iiGPU::iiGPU iiGPU::WebGPU iiWebGPU::iiWebGPU)
    add_test(NAME installed.webgpu_memory COMMAND consumer)
elseif(WITH_CORE_COMPOSITION)
    find_package(iiGPU 0.1 CONFIG REQUIRED COMPONENTS Runtime)
    find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
    add_executable(consumer ../completion.cpp)
    target_link_libraries(consumer PRIVATE iiGPU::Runtime iiSDKCore::Runtime)
    add_test(NAME installed.core_composition COMMAND consumer)
elseif(WITH_RUNTIME)
    find_package(iiGPU 0.1 CONFIG REQUIRED COMPONENTS Runtime)
    if(TARGET iiSDKCore::Runtime)
        message(FATAL_ERROR "Completion observation must not import the Core runtime")
    endif()
    add_executable(consumer runtime.cpp)
    target_link_libraries(consumer PRIVATE iiGPU::Runtime)
    add_test(NAME installed.runtime COMMAND consumer)
else()
    find_package(iiGPU 0.1 CONFIG REQUIRED)
    if(TARGET iiSDKCore::Runtime)
        message(FATAL_ERROR "Base-only consumption must not import the Core runtime")
    endif()
    add_executable(consumer main.cpp)
    target_link_libraries(consumer PRIVATE iiGPU::iiGPU)
    add_test(NAME installed.buffer COMMAND consumer)
    add_executable(installed_example "${iiGPU_EXAMPLE_DIR}/main.cpp")
    target_link_libraries(installed_example PRIVATE iiGPU::iiGPU)
    foreach(_api metal vulkan)
        add_test(NAME installed.${_api}.compute COMMAND installed_example ${_api})
    endforeach()
endif()
get_property(_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${_tests} PROPERTIES TIMEOUT 30)
