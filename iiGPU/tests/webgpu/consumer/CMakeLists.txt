cmake_minimum_required(VERSION 3.31)
project(iiWebGPUInstalledConsumer LANGUAGES CXX)
include(CTest)
find_package(iiGPU 0.1 CONFIG REQUIRED COMPONENTS WebGPU)
find_package(iiWebGPU 0.1 CONFIG REQUIRED)
if(TARGET Qt6::Core OR TARGET Qt5::Core OR TARGET iiSDKCore::iiSDKCore)
    message(FATAL_ERROR "WebGPU consumer imported an unrelated runtime")
endif()
add_executable(iiWebGPUInstalledConsumer ../../../example/WebGPU/compute.cpp)
target_link_libraries(iiWebGPUInstalledConsumer PRIVATE iiGPU::WebGPU iiWebGPU::iiWebGPU)
add_test(NAME iiWebGPU.InstalledCompute COMMAND iiWebGPUInstalledConsumer)
