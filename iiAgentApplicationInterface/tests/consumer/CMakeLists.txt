cmake_minimum_required(VERSION 3.24)
project(iiAgentApplicationInterfaceInstalledConsumer LANGUAGES CXX)
find_package(iiAgentApplicationInterface CONFIG REQUIRED)
if(NOT EXISTS "${iiAgentApplicationInterface_EXAMPLE_DIR}/iiNetworkExternalBridge.cpp")
    message(FATAL_ERROR "Installed app-side bridge path is unavailable")
endif()
add_executable(consumer main.cpp)
target_include_directories(consumer BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(consumer PRIVATE iiAgentApplicationInterface::iiAgentApplicationInterface)
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
