cmake_minimum_required(VERSION 3.24)
project(BuildDiscoveryConsumer LANGUAGES CXX)
find_package(iiAgentApplicationInterface CONFIG REQUIRED)
add_executable(discovery_app "${TEST_RESOURCES}/main.cpp" "${TEST_RESOURCES}/model.cpp")
target_include_directories(discovery_app PRIVATE "${TEST_RESOURCES}" "${TEST_RESOURCES}/priority_user")
target_include_directories(discovery_app SYSTEM PRIVATE "${TEST_RESOURCES}/priority_system")
add_library(discovery_component STATIC "${TEST_RESOURCES}/component.cpp")
target_link_libraries(discovery_app PRIVATE discovery_component)
option(TEST_DISCOVERY_EXTRA "Enable a conditional application member" OFF)
if(TEST_DISCOVERY_EXTRA)
    target_compile_definitions(discovery_app PRIVATE BUILD_DISCOVERY_TEST_EXTRA=1)
endif()
iiAgentApplicationInterface_enable_discovery(discovery_app)
enable_testing()
add_test(NAME generated_catalog COMMAND discovery_app)
