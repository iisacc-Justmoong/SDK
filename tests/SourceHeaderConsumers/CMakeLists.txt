cmake_minimum_required(VERSION 3.31)
project(SourceHeaderConsumers LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/source-header-layout.json" _layout)
string(JSON _count LENGTH "${_layout}" modules)
math(EXPR _last "${_count} - 1")
foreach(_index RANGE 0 ${_last})
    string(JSON _module GET "${_layout}" modules ${_index} module)
    string(JSON _header_count LENGTH "${_layout}" modules ${_index} publicHeaders)
    math(EXPR _header_last "${_header_count} - 1")
    set(_checks)
    foreach(_header_index RANGE 0 ${_header_last})
        string(JSON _source_header GET "${_layout}" modules ${_index} publicHeaders ${_header_index})
        if(NOT _source_header MATCHES "^src/")
            continue()
        endif()
        string(REGEX REPLACE "^src/" "" _public_header "${_source_header}")
        set(_check "${CMAKE_CURRENT_BINARY_DIR}/headers/${_module}-${_header_index}.cpp")
        file(WRITE "${_check}" "#include <${_public_header}>\n")
        list(APPEND _checks "${_check}")
    endforeach()
    if(NOT _checks)
        continue()
    endif()
    find_package(${_module} CONFIG REQUIRED)
    if(DEFINED HEADER_STAGE_ROOT)
        cmake_path(IS_PREFIX HEADER_STAGE_ROOT "${${_module}_DIR}" NORMALIZE _staged)
        if(NOT _staged)
            message(FATAL_ERROR "${_module} was not selected from the staged packages: ${${_module}_DIR}")
        endif()
    endif()
    set(_main "${CMAKE_CURRENT_BINARY_DIR}/headers/${_module}-main.cpp")
    file(WRITE "${_main}" "int main() { return 0; }\n")
    add_executable(${_module}HeaderConsumer "${_main}" ${_checks})
    target_compile_features(${_module}HeaderConsumer PRIVATE cxx_std_23)
    target_link_libraries(${_module}HeaderConsumer PRIVATE ${_module}::${_module})
    add_test(NAME SourceHeaderConsumer.${_module} COMMAND ${_module}HeaderConsumer)
endforeach()
