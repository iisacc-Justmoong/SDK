cmake_minimum_required(VERSION 3.24)
project(iiSDKQtFreeConsumer VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
option(IISDK_QTFREE_SOURCE_MODULES "Compile the changed modules instead of finding their installed packages" OFF)
option(IISDK_QTFREE_SANITIZERS "Instrument the changed modules and consumers" OFF)
if(IISDK_QTFREE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(iiSDKCore 0.18 CONFIG REQUIRED COMPONENTS Runtime)
find_package(iiJson 0.1 CONFIG REQUIRED)
if(IISDK_QTFREE_SOURCE_MODULES)
    set(IINETWORK_BUILD_TESTS ON CACHE BOOL "" FORCE)
    set(IINETWORK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(../../iiFileProvider/Native provider)
    add_subdirectory(../../iiImageCodec image)
    add_subdirectory(../../iiXml/Native xml)
    add_subdirectory(../../iiNetwork network)
else()
    find_package(iiFileProviderNative 0.9 CONFIG REQUIRED)
    find_package(iiImageCodec 0.1 CONFIG REQUIRED)
    find_package(iiXmlNative 0.1 CONFIG REQUIRED)
    find_package(iiNetwork 0.5 CONFIG REQUIRED)
endif()
if(TARGET Qt6::Core OR TARGET Qt5::Core)
    message(FATAL_ERROR "The non-GUI consumer imported Qt")
endif()
add_executable(iiSDKQtFreeConsumer main.cpp)
target_compile_features(iiSDKQtFreeConsumer PRIVATE cxx_std_23)
target_link_libraries(iiSDKQtFreeConsumer PRIVATE iiSDKCore::Runtime iiJson::iiJson
    iiFileProvider::Native iiImageCodec::iiImageCodec iiXml::Native iiNetwork::iiNetwork)
add_test(NAME iiSDK.QtFree.Consumer COMMAND iiSDKQtFreeConsumer "${CMAKE_CURRENT_BINARY_DIR}")
if(APPLE)
    set_tests_properties(iiSDK.QtFree.Consumer PROPERTIES ENVIRONMENT_MODIFICATION
        "DYLD_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_VERSIONED_LIBRARY_PATH=unset:")
endif()
find_path(IILOCALDIFFUSION_INCLUDE_DIRECTORY Generation/NativeDiffusion.hpp)
if(IILOCALDIFFUSION_INCLUDE_DIRECTORY)
    add_executable(iiSDKQtFreeGeneration ../../../Product/Dreamscapes/tests/tst_nativegeneration.cpp
        ../../../Product/Dreamscapes/src/App/Generation/NativeGenerationWork.cpp)
    target_compile_features(iiSDKQtFreeGeneration PRIVATE cxx_std_23)
    target_include_directories(iiSDKQtFreeGeneration PRIVATE
        ../../../Product/Dreamscapes/src/App/Generation "${IILOCALDIFFUSION_INCLUDE_DIRECTORY}")
    target_link_libraries(iiSDKQtFreeGeneration PRIVATE iiFileProvider::Native iiImageCodec::iiImageCodec)
    add_test(NAME iiSDK.QtFree.Generation COMMAND iiSDKQtFreeGeneration "${CMAKE_CURRENT_BINARY_DIR}")
    if(APPLE)
        set_tests_properties(iiSDK.QtFree.Generation PROPERTIES ENVIRONMENT_MODIFICATION
            "DYLD_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_VERSIONED_LIBRARY_PATH=unset:")
    endif()
endif()
