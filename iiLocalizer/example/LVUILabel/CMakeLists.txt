cmake_minimum_required(VERSION 3.24)
project(iiLocalizerLVUILabelConsumer LANGUAGES CXX)
include(CTest)
get_filename_component(II_LOCALIZER_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IIGUI_SOURCE_DIR "${II_LOCALIZER_SOURCE_ROOT}/../iiGUI" CACHE PATH "iiGUI source used only by the headless test harness")
find_package(iiLocalizer CONFIG REQUIRED)
find_package(iiGUI CONFIG REQUIRED)
find_package(LVUI CONFIG REQUIRED COMPONENTS Native)
find_package(iiSDKCore CONFIG REQUIRED COMPONENTS Runtime)
lvui_generate(NAME LocalizedLabel SOURCE "${II_LOCALIZER_SOURCE_ROOT}/resources/tests/LocalizedLabel.lvui" OUTPUT_VARIABLE generated_view)
add_executable(iiLocalizerLVUILabel main.cpp "${generated_view}")
target_compile_features(iiLocalizerLVUILabel PRIVATE cxx_std_23)
target_include_directories(iiLocalizerLVUILabel PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${IIGUI_SOURCE_DIR}/src/Widgets")
target_compile_definitions(iiLocalizerLVUILabel PRIVATE "IILOCALIZER_TEST_PO_PATH=\"${II_LOCALIZER_SOURCE_ROOT}/resources/tests/ko.po\"")
iilocalizer_link(iiLocalizerLVUILabel)
target_link_libraries(iiLocalizerLVUILabel PRIVATE LVUI::Native iiGUI::iiGUI iiSDKCore::Runtime)
add_test(NAME iiLocalizer.LVUILabel COMMAND iiLocalizerLVUILabel)
